# Build-zone dependencies that come from PIP.
#
# The compiled scientific stack is deliberately NOT here — numpy, scipy, librosa,
# soundfile and scikit-learn are installed from conda-forge in pipeline.Dockerfile,
# alongside MFA, so they share one toolchain and one libstdc++.
#
# Installing them by pip into a conda base produced this at image build time:
#
#     ImportError: /lib/x86_64-linux-gnu/libstdc++.so.6: version `CXXABI_1.3.15' not found
#                  (required by .../scipy/spatial/_distance_pybind...so)
#
# because pip's wheels are built against a newer libstdc++ than Ubuntu jammy ships, and the
# system copy wins the linker search. If you add a compiled package, put it in the conda
# line, not here.
#
# torch/torchaudio are also absent: pipeline.Dockerfile installs them first from the
# CPU-only index so speechbrain cannot pull the CUDA build in as a transitive dependency.

requests>=2.31
pypdf>=4.0
boto3>=1.34            # S3-compatible object storage

# ECAPA speaker embeddings. The >=1.1 floor is NOT cosmetic.
#
# speechbrain 1.0.3 — which `>=1.0` resolved to, and which plexus verified against an
# older stack — calls huggingface_hub.hf_hub_download(use_auth_token=...). That argument
# was removed in huggingface_hub 1.x, and the MFA base image ships 1.23.0, so 1.0.3 dies
# with:
#
#     TypeError: hf_hub_download() got an unexpected keyword argument 'use_auth_token'
#
# This is NOT only a build-time problem. pipeline/speech/diarize.py:127 calls
# EncoderClassifier.from_hparams() at RUNTIME, so the embed stage would fail on a pod
# exactly the same way. Pinning huggingface_hub<1.0 instead was rejected: MFA's own
# models/classes.py imports huggingface_hub, so dragging it backwards risks breaking
# alignment to fix embedding.
#
# speechbrain 1.1.0 uses token= and fetches ECAPA cleanly against hf_hub 1.23 (verified).
speechbrain>=1.1
